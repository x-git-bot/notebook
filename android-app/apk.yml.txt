name: Build Android APK

on:
  workflow_dispatch:
  push:
    paths:
      - 'index.html'
      - 'template.html'
      - 'android-app/**'
      - '.github/workflows/apk.yml'

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Copy the web app into the Android project
        working-directory: android-app
        run: |
          rm -rf www && mkdir -p www
          cp ../index.html ../template.html ../manifest.json www/
          cp -r ../icons www/icons

      - name: Install
        working-directory: android-app
        run: npm install

      - name: Create Android project
        working-directory: android-app
        run: |
          npx cap add android
          npx cap sync android

      - name: App icon and splash (optional)
        working-directory: android-app
        continue-on-error: true
        run: npx @capacitor/assets generate --android --iconBackgroundColor "#090a1c" --splashBackgroundColor "#090a1c"

      - name: Use the fixed signing key (so updates install over the old app)
        run: |
          mkdir -p ~/.android
          cp android-app/debug.keystore ~/.android/debug.keystore

      - name: Build APK
        working-directory: android-app/android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: Notes-APK
          path: android-app/android/app/build/outputs/apk/debug/app-debug.apk
